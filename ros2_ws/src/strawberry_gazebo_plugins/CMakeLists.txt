cmake_minimum_required(VERSION 3.16)
project(strawberry_gazebo_plugins LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
add_library(strawberry_lazy_detachable_joint SHARED src/LazyDetachableJoint.cc)
target_compile_features(strawberry_lazy_detachable_joint PRIVATE cxx_std_17)
target_include_directories(strawberry_lazy_detachable_joint PRIVATE include)
target_link_libraries(strawberry_lazy_detachable_joint PRIVATE gz-sim8::gz-sim8)
install(TARGETS strawberry_lazy_detachable_joint LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_attachment_policy test/test_attachment_policy.cc)
  target_include_directories(test_attachment_policy PRIVATE include)
  target_compile_features(test_attachment_policy PRIVATE cxx_std_17)
  add_test(NAME attachment_policy COMMAND test_attachment_policy)
endif()
ament_package()
